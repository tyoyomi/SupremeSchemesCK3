ssc_bestow_supreme_schemer_interaction = {
	category = interaction_category_friendly
	icon = intrigue
	interface_priority = 55
	common_interaction = yes

	desc = ssc_bestow_supreme_schemer_interaction_desc

	auto_accept = yes

	is_shown = {
		has_game_rule = ssc_rule_granting_player_and_targets
		scope:actor = {
			is_alive = yes
			ssc_actor_may_bestow_by_rule = yes
		}
		scope:recipient = {
			is_alive = yes
			is_adult = yes
			NOT = { has_trait = ssc_supreme_schemer }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = ssc_tt_actor_may_bestow
				ssc_actor_may_bestow_by_rule = yes
			}
		}
		scope:recipient = {
			custom_tooltip = {
				text = ssc_tt_can_receive_schemer
				is_adult = yes
				NOT = { has_trait = ssc_supreme_schemer }
			}
		}
	}

	on_accept = {
		scope:recipient = {
			ssc_bestow_supreme_schemer_effect = yes
		}
	}

	ai_targets = {
		ai_recipients = courtiers
		chance = 0
	}
	ai_frequency = 0

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
