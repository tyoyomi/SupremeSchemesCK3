		limit = { has_game_rule = ssc_rule_granting_player_self }

	}

		limit = { has_game_rule = ssc_rule_granting_player_and_targets }
		OR = {
			is_ai = no
			has_game_rule = ssc_rule_ai_follow_intensity
		}
	}
	trigger_else = {
		always = no
	}
}

# Cap values: 1=county, 2=duchy, 3=kingdom, 4=empire
ssc_title_cap_allows_duchy = {
	OR = {
		AND = {
			exists = var:ssc_cfg_title_cap
			var:ssc_cfg_title_cap >= 2
		}
		AND = {
			NOT = { exists = var:ssc_cfg_title_cap }
			OR = {
				has_game_rule = ssc_rule_title_steal_cap_duchy
				has_game_rule = ssc_rule_title_steal_cap_kingdom
				has_game_rule = ssc_rule_title_steal_cap_empire
			}
		}
	}
	ssc_intensity_is_high = yes
}

ssc_title_cap_allows_kingdom = {
	OR = {
		AND = {
			exists = var:ssc_cfg_title_cap
			var:ssc_cfg_title_cap >= 3
		}
		AND = {
			NOT = { exists = var:ssc_cfg_title_cap }
			OR = {
				has_game_rule = ssc_rule_title_steal_cap_kingdom
				has_game_rule = ssc_rule_title_steal_cap_empire

ssc_title_cap_allows_empire = {

# this = title; safe to steal via intrigue
ssc_title_is_stealable = {
	exists = holder
	is_landless_type_title = no
	is_noble_family_title = no
	NOT = { tier = tier_barony }
	holder = {
		is_alive = yes
		is_landed = yes
		NOT = { this = scope:owner }
		NOT = { government_has_flag = government_is_holy_order }
	}
	# Mid: county only. High: up to configured cap (county→empire).
	trigger_if = {
		limit = {
			exists = scope:owner
			scope:owner = { ssc_intensity_is_mid = yes }
		}
		tier = tier_county
	}
	trigger_else_if = {
		limit = {
			exists = scope:owner
			scope:owner = { ssc_title_cap_allows_empire = yes }
		}
		OR = {
			tier = tier_county
			tier = tier_duchy
			tier = tier_kingdom
			tier = tier_empire
		}
	}
	trigger_else_if = {
		limit = {
			exists = scope:owner
			scope:owner = { ssc_title_cap_allows_kingdom = yes }
		}
		OR = {
			tier = tier_county
			tier = tier_duchy
			tier = tier_kingdom
		}
	}
	trigger_else_if = {
		limit = {
			exists = scope:owner
			scope:owner = { ssc_title_cap_allows_duchy = yes }
		}
		OR = {
			tier = tier_county
			tier = tier_duchy
		}
	}
	trigger_else = {
		tier = tier_county
	}
}

# this = title; leave target landed after steal
ssc_title_steal_leaves_holder_landed = {
	holder = {
		save_temporary_scope_as = ssc_title_holder
	}
	OR = {
		tier = tier_duchy
		tier = tier_kingdom
		tier = tier_empire
		AND = {
			tier = tier_county
			scope:ssc_title_holder = {
				any_held_title = {
					count >= 2
					tier = tier_county
				}
			}
		}
	}
	# Kingdom/empire steals require the holder keep at least one county
	trigger_if = {
		limit = {
			OR = {
				tier = tier_kingdom
				tier = tier_empire
			}
		}
		scope:ssc_title_holder = {
			any_held_title = {
				tier = tier_county
			}
		}
	}
}

# this = reluctant marriage target; expects scope:owner = schemer
ssc_force_contract_has_eligible_match = {
	save_temporary_scope_as = ssc_contract_target
	is_married = no
	exists = scope:owner
	OR = {
		scope:owner = {
			is_married = no
			can_marry_character = scope:ssc_contract_target
			scope:ssc_contract_target = { can_marry_character = prev }
		}
		scope:owner = {
			any_courtier = {
				is_alive = yes
				is_adult = yes
				is_married = no
				can_marry_character = scope:ssc_contract_target
				scope:ssc_contract_target = { can_marry_character = prev }
			}
		}
		scope:owner = {
			any_close_family_member = {
				is_alive = yes
				is_adult = yes
				is_married = no
				can_marry_character = scope:ssc_contract_target
				scope:ssc_contract_target = { can_marry_character = prev }
			}
		}
	}
}
