# Supreme Schemes — scripted triggers

ssc_is_supreme_schemer = {
	has_trait = ssc_supreme_schemer
}

# Soft peer: Supreme Sovereign trait key. Fails closed if SS is absent (may log once).
ssc_character_is_supreme_sovereign = {
	has_trait = Supreme_Sovereign
}

ssc_can_demand_strong_hook_actor = {
	is_alive = yes
	is_ai = no
	ssc_character_is_supreme_sovereign = yes
}

# Player Supreme Schemer and/or Supreme Sovereign: bypass hook spend + fabricate-hook CDs
ssc_player_bypasses_hook_cooldowns = {
	is_alive = yes
	is_ai = no
	OR = {
		has_trait = ssc_supreme_schemer
		ssc_character_is_supreme_sovereign = yes
	}
}

ssc_can_receive_strong_hook_demand = {
	is_alive = yes
	NOT = { this = scope:actor }
}

ssc_granting_disabled = {
	has_game_rule = ssc_rule_granting_disabled
}

ssc_can_embrace_schemes = {
	is_alive = yes
	is_adult = yes
	NOT = { has_game_rule = ssc_rule_granting_disabled }
	NOT = { has_trait = ssc_supreme_schemer }
	trigger_if = {
		limit = { has_game_rule = ssc_rule_granting_player_self }
		is_ai = no
	}
	trigger_else_if = {
		limit = { has_game_rule = ssc_rule_granting_player_and_targets }
		OR = {
			is_ai = no
			has_game_rule = ssc_rule_ai_follow_intensity
		}
	}
	trigger_else = {
		always = no
	}
}

ssc_module_turbo_on = {
	OR = {
		AND = {
			exists = var:ssc_cfg_module_turbo
			var:ssc_cfg_module_turbo = 1
		}
		AND = {
			NOT = { exists = var:ssc_cfg_module_turbo }
			has_game_rule = ssc_rule_module_turbo_on
		}
	}
	NOT = {
		AND = {
			exists = var:ssc_cfg_module_turbo
			var:ssc_cfg_module_turbo = 0
		}
	}
}

ssc_intensity_is_low = {
	exists = var:ssc_cfg_intensity
	var:ssc_cfg_intensity = 1
}

ssc_intensity_is_mid = {
	exists = var:ssc_cfg_intensity
	var:ssc_cfg_intensity = 2
}

ssc_intensity_is_high = {
	OR = {
		NOT = { exists = var:ssc_cfg_intensity }
		var:ssc_cfg_intensity = 3
	}
}

# Cap values: 1=county, 2=duchy, 3=kingdom, 4=empire
ssc_title_cap_allows_duchy = {
	OR = {
		AND = {
			exists = var:ssc_cfg_title_cap
			var:ssc_cfg_title_cap >= 2
		}
		AND = {
			NOT = { exists = var:ssc_cfg_title_cap }
			OR = {
				has_game_rule = ssc_rule_title_steal_cap_duchy
				has_game_rule = ssc_rule_title_steal_cap_kingdom
				has_game_rule = ssc_rule_title_steal_cap_empire
			}
		}
	}
	ssc_intensity_is_high = yes
}

ssc_title_cap_allows_kingdom = {
	OR = {
		AND = {
			exists = var:ssc_cfg_title_cap
			var:ssc_cfg_title_cap >= 3
		}
		AND = {
			NOT = { exists = var:ssc_cfg_title_cap }
			OR = {
				has_game_rule = ssc_rule_title_steal_cap_kingdom
				has_game_rule = ssc_rule_title_steal_cap_empire
			}
		}
	}
	ssc_intensity_is_high = yes
}

ssc_title_cap_allows_empire = {
	OR = {
		AND = {
			exists = var:ssc_cfg_title_cap
			var:ssc_cfg_title_cap >= 4
		}
		AND = {
			NOT = { exists = var:ssc_cfg_title_cap }
			has_game_rule = ssc_rule_title_steal_cap_empire
		}
	}
	ssc_intensity_is_high = yes
}

# this = title; safe to steal via intrigue
ssc_title_is_stealable = {
	exists = holder
	is_landless_type_title = no
	is_noble_family_title = no
	NOT = { tier = tier_barony }
	holder = {
		is_alive = yes
		is_landed = yes
		NOT = { this = scope:owner }
		NOT = { government_has_flag = government_is_holy_order }
	}
	# Mid: county only. High: up to configured cap (county→empire).
	trigger_if = {
		limit = {
			exists = scope:owner
			scope:owner = { ssc_intensity_is_mid = yes }
		}
		tier = tier_county
	}
	trigger_else_if = {
		limit = {
			exists = scope:owner
			scope:owner = { ssc_title_cap_allows_empire = yes }
		}
		OR = {
			tier = tier_county
			tier = tier_duchy
			tier = tier_kingdom
			tier = tier_empire
		}
	}
	trigger_else_if = {
		limit = {
			exists = scope:owner
			scope:owner = { ssc_title_cap_allows_kingdom = yes }
		}
		OR = {
			tier = tier_county
			tier = tier_duchy
			tier = tier_kingdom
		}
	}
	trigger_else_if = {
		limit = {
			exists = scope:owner
			scope:owner = { ssc_title_cap_allows_duchy = yes }
		}
		OR = {
			tier = tier_county
			tier = tier_duchy
		}
	}
	trigger_else = {
		tier = tier_county
	}
}

# this = title; leave target landed after steal
ssc_title_steal_leaves_holder_landed = {
	holder = {
		save_temporary_scope_as = ssc_title_holder
	}
	OR = {
		tier = tier_duchy
		tier = tier_kingdom
		tier = tier_empire
		AND = {
			tier = tier_county
			scope:ssc_title_holder = {
				any_held_title = {
					count >= 2
					tier = tier_county
				}
			}
		}
	}
	# Kingdom/empire steals require the holder keep at least one county
	trigger_if = {
		limit = {
			OR = {
				tier = tier_kingdom
				tier = tier_empire
			}
		}
		scope:ssc_title_holder = {
			any_held_title = {
				tier = tier_county
			}
		}
	}
}

ssc_actor_may_bestow_by_rule = {
	NOT = { has_game_rule = ssc_rule_granting_disabled }
	has_game_rule = ssc_rule_granting_player_and_targets
	trigger_if = {
		limit = { is_ai = yes }
		has_game_rule = ssc_rule_ai_follow_intensity
	}
}

ssc_ai_may_use_ssc_schemes = {
	OR = {
		is_ai = no
		has_game_rule = ssc_rule_ai_follow_intensity
	}
}
