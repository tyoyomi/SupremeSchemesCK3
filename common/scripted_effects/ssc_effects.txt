# Supreme Schemes — scripted effects
# Bestow expects this = recipient. Scheme outcomes expect scope:owner / scope:target / scope:scheme.

### Defaults / refresh / cycle
ssc_ensure_schemer_defaults_effect = {
	if = {
		limit = { NOT = { exists = var:ssc_cfg_intensity } }
		if = {
			limit = { has_game_rule = ssc_rule_intensity_default_low }
			set_variable = { name = ssc_cfg_intensity value = 1 }
		}
		else_if = {
			limit = { has_game_rule = ssc_rule_intensity_default_mid }
			set_variable = { name = ssc_cfg_intensity value = 2 }
		}
		else = {
			set_variable = { name = ssc_cfg_intensity value = 3 }
		}
	}
	if = {
		limit = { NOT = { exists = var:ssc_cfg_module_turbo } }
		if = {
			limit = { has_game_rule = ssc_rule_module_turbo_off }
			set_variable = { name = ssc_cfg_module_turbo value = 0 }
		}
		else = {
			set_variable = { name = ssc_cfg_module_turbo value = 1 }
		}
	}
	if = {
		limit = { NOT = { exists = var:ssc_cfg_module_artifact } }
		if = {
			limit = { has_game_rule = ssc_rule_module_artifact_theft_off }
			set_variable = { name = ssc_cfg_module_artifact value = 0 }
		}
		else = {
			set_variable = { name = ssc_cfg_module_artifact value = 1 }
		}
	}
	if = {
		limit = { NOT = { exists = var:ssc_cfg_module_title } }
		if = {
			limit = { has_game_rule = ssc_rule_module_title_intrigue_off }
			set_variable = { name = ssc_cfg_module_title value = 0 }
		}
		else = {
			set_variable = { name = ssc_cfg_module_title value = 1 }
		}
	}
	if = {
		limit = { NOT = { exists = var:ssc_cfg_module_auto_hooks } }
		if = {
			limit = { has_game_rule = ssc_rule_module_auto_hooks_off }
			set_variable = { name = ssc_cfg_module_auto_hooks value = 0 }
		}
		else = {
			set_variable = { name = ssc_cfg_module_auto_hooks value = 1 }
		}
	}
	if = {
		limit = { NOT = { exists = var:ssc_cfg_title_cap } }
		if = {
			limit = { has_game_rule = ssc_rule_title_steal_cap_county }
			set_variable = { name = ssc_cfg_title_cap value = 1 }
		}
		else_if = {
			limit = { has_game_rule = ssc_rule_title_steal_cap_empire }
			set_variable = { name = ssc_cfg_title_cap value = 4 }
		}
		else_if = {
			limit = { has_game_rule = ssc_rule_title_steal_cap_kingdom }
			set_variable = { name = ssc_cfg_title_cap value = 3 }
		}
		else = {
			set_variable = { name = ssc_cfg_title_cap value = 2 }
		}
	}
}

ssc_clear_schemer_modifiers_effect = {
	remove_character_modifier = ssc_turbo_low_modifier
	remove_character_modifier = ssc_turbo_med_modifier
	remove_character_modifier = ssc_turbo_high_modifier
}

ssc_refresh_schemer_modifiers_effect = {
	ssc_clear_schemer_modifiers_effect = yes
	ssc_ensure_schemer_defaults_effect = yes
	if = {
		limit = {
			has_trait = ssc_supreme_schemer
			ssc_module_turbo_on = yes
		}
		if = {
			limit = { var:ssc_cfg_intensity = 1 }
			add_character_modifier = {
				modifier = ssc_turbo_low_modifier
				days = -1
			}
		}
		else_if = {
			limit = { var:ssc_cfg_intensity = 2 }
			add_character_modifier = {
				modifier = ssc_turbo_med_modifier
				days = -1
			}
		}
		else = {
			add_character_modifier = {
				modifier = ssc_turbo_high_modifier
				days = -1
			}
		}
	}
}

# this = schemer discovering secrets about scope:ssc_secret_target
ssc_discover_all_secrets_of_target_effect = {
	if = {
		limit = {
			exists = scope:ssc_secret_target
			scope:ssc_secret_target = { is_alive = yes }
		}
		save_scope_as = ssc_discoverer
		scope:ssc_secret_target = {
			every_secret = {
				limit = {
					NOT = { is_known_by = scope:ssc_discoverer }
				}
				reveal_to = scope:ssc_discoverer
			}
		}
	}
}

### Artifact pick / transfer
ssc_pick_default_artifact_effect = {
	# root/scheme context: expects scope:scheme and scope:target
	scope:target = {
		random_character_artifact = {
			weight = {
				base = 1
				modifier = {
					add = 100
					rarity = illustrious
				}
				modifier = {
					add = 50
					rarity = famed
				}
				modifier = {
					add = 25
					rarity = masterwork
				}
			}
			save_scope_as = artifact
		}
	}
	if = {
		limit = { exists = scope:artifact }
		scope:scheme = {
			set_variable = {
				name = target_artifact
				value = scope:artifact
			}
		}
	}
}

ssc_transfer_stolen_artifact_effect = {
	# expects scope:owner, scope:target, scope:artifact
	if = {
		limit = {
			exists = scope:artifact
			exists = scope:owner
			exists = scope:target
		}
		steal_artifact_in_event_effect = {
			THIEF = scope:owner
			VICTIM = scope:target
			ARTIFACT = scope:artifact
		}
		scope:target = {
			add_opinion = {
				target = scope:owner
				modifier = stole_artifact_from_me
			}
		}
		scope:owner = {
			ssc_apply_steal_artifact_cooldown_effect = yes
			trigger_event = ssc_events.2001
		}
	}
}

ssc_apply_fabricated_claim_effect = {
	# expects scope:owner, scope:target (title)
	if = {
		limit = {
			exists = scope:owner
			exists = scope:target
		}
		if = {
			limit = { scope:owner = { ssc_intensity_is_low = yes } }
			scope:owner = {
				add_unpressed_claim = scope:target
			}
		}
		else = {
			scope:owner = {
				add_pressed_claim = scope:target
			}
		}
		if = {
			limit = { exists = scope:target.holder }
			scope:target.holder = {
				add_opinion = {
					target = scope:owner
					modifier = ssc_fabricated_claim_opinion
				}
			}
		}
		scope:owner = {
			ssc_apply_fabricate_claim_cooldown_effect = yes
			trigger_event = ssc_events.3001
		}
	}
}

ssc_transfer_stolen_title_effect = {
	# expects scope:owner, scope:target (title)
	if = {
		limit = {
			exists = scope:owner
			exists = scope:target
			scope:target = {
				ssc_title_is_stealable = yes
				ssc_title_steal_leaves_holder_landed = yes
			}
		}
		scope:target.holder = { save_scope_as = ssc_former_holder }
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = ssc_title_change
			add_claim_on_loss = yes
		}
		scope:target = {
			change_title_holder = {
				holder = scope:owner
				change = scope:ssc_title_change
			}
		}
		resolve_title_and_vassal_change = scope:ssc_title_change
		if = {
			limit = { exists = scope:ssc_former_holder }
			scope:ssc_former_holder = {
				add_opinion = {
					target = scope:owner
					modifier = ssc_stolen_title_opinion
				}
			}
		}
		scope:owner = {
			ssc_apply_steal_title_cooldown_effect = yes
			trigger_event = ssc_events.3101
		}
	}
}

### Scheme ending helpers
ssc_scheme_roll_success_effect = {
	# root = scheme; sets scope:scheme_successful on success
	if = {
		limit = {
			scheme_owner = { ssc_intensity_is_high = yes }
		}
		save_scope_value_as = {
			name = scheme_successful
			value = yes
		}
	}
	else = {
		random = {
			chance = scheme_success_chance
			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}
	}
}

ssc_steal_artifact_scheme_ending_effect = {
	save_scope_as = scheme
	scheme_owner = { save_scope_as = owner }
	scheme_target_character = { save_scope_as = target }
	if = {
		limit = {
			exists = var:target_artifact
		}
		var:target_artifact = { save_scope_as = artifact }
	}
	else = {
		ssc_pick_default_artifact_effect = yes
	}
	ssc_scheme_roll_success_effect = yes
	if = {
		limit = { exists = scope:scheme_successful }
		ssc_transfer_stolen_artifact_effect = yes
	}
	else = {
		scope:target = {
			add_opinion = {
				target = scope:owner
				modifier = ssc_attempted_scheme_opinion
			}
			add_character_modifier = {
				modifier = watchful_modifier
				days = 365
			}
		}
		scope:owner = {
			ssc_apply_steal_artifact_cooldown_effect = yes
			trigger_event = ssc_events.2002
		}
	}
	end_scheme = yes
}

### Strong favor hook apply (shared)
# this = hook owner; expects scope:ssc_hook_target.
# Optional: save_scope_value_as = { name = ssc_hook_notify value = yes } before call for toast.
ssc_apply_strong_favor_hook_effect = {
	if = {
		limit = {
			exists = scope:ssc_hook_target
			scope:ssc_hook_target = { is_alive = yes }
			can_add_hook = {
				type = strong_favor_hook
				target = scope:ssc_hook_target
			}
		}
		add_hook = {
			type = strong_favor_hook
			target = scope:ssc_hook_target
		}
		if = {
			limit = {
				exists = scope:ssc_hook_notify
				scope:ssc_hook_notify = yes
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = ssc_demand_strong_hook_toast_title
				left_icon = scope:ssc_hook_target
				custom_tooltip = ssc_demand_strong_hook_toast_tt
			}
		}
	}
}

### Strong hook demand (player Supreme Sovereign -> recipient)

### Auto Strong Hooks — silent try on new vassal
# root/this = new liege; scope:vassal = gained vassal. No toast (ASH11).
ssc_try_auto_hook_new_vassal_effect = {
	if = {
		limit = {
			ssc_can_auto_strong_hook_liege = yes
			exists = scope:vassal
		}
		# Force silent (no per-vassal toast)
		save_scope_value_as = {
			name = ssc_hook_notify
			value = no
		}
		save_scope_as = ssc_hook_owner
		scope:vassal = {
			if = {
				limit = { ssc_can_receive_auto_strong_hook = yes }
				save_scope_as = ssc_hook_target
				scope:ssc_hook_owner = {
					ssc_apply_strong_favor_hook_effect = yes
				}
			}
		}
	}
}

### Hook All Vassals — mass silent apply + one summary toast
# this = player liege. Direct vassals only. No cooldown. Does not require module On.
ssc_hook_all_direct_vassals_effect = {
	save_scope_as = ssc_hook_owner
	# Force silent per-vassal apply; summary toast is separate
	save_scope_value_as = {
		name = ssc_hook_notify
		value = no
	}
	set_variable = {
		name = ssc_hook_all_count
		value = 0
	}
	every_vassal = {
		limit = { ssc_can_receive_auto_strong_hook = yes }
		save_scope_as = ssc_hook_target
		scope:ssc_hook_owner = {
			ssc_apply_strong_favor_hook_effect = yes
			change_variable = {
				name = ssc_hook_all_count
				add = 1
			}
		}
	}
	if = {
		limit = { var:ssc_hook_all_count > 0 }
		send_interface_toast = {
			type = event_toast_effect_good
			title = ssc_hook_all_vassals_toast_title
			left_icon = root
			custom_tooltip = ssc_hook_all_vassals_toast_tt
		}
	}
	# Keep ssc_hook_all_count so the toast tooltip can still resolve the count.
}

### Hook cooldown bypass (player schemer and/or Sovereign)
# on_hook_used: root = user, scope:target = hooked character.
# Strong hooks stay but go on HOOK_COOLDOWN_DURATION_YEARS; remove+readd clears CD.
ssc_refresh_hook_after_use_effect = {
	if = {
		limit = {
			ssc_player_bypasses_hook_cooldowns = yes
			exists = scope:target
			scope:target = { is_alive = yes }
			has_strong_hook = scope:target
			NOT = { has_strong_usable_hook = scope:target }
		}
		# Preserve type when known; otherwise fall back to strong favor.
		if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = strong_favor_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = strong_favor_hook target = scope:target }
		}
		else_if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = strong_blackmail_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = strong_blackmail_hook target = scope:target }
		}
		else_if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = strong_obligation_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = strong_obligation_hook target = scope:target }
		}
		else_if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = strong_indebted_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = strong_indebted_hook target = scope:target }
		}
		else_if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = fabrication_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = fabrication_hook target = scope:target }
		}
		else_if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = life_threat_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = life_threat_hook target = scope:target }
		}
		else_if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = major_crime_accomplice_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = major_crime_accomplice_hook target = scope:target }
		}
		else_if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = loyalty_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = loyalty_hook target = scope:target }
		}
		else_if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = strong_influence_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = strong_influence_hook target = scope:target }
		}
		else_if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = liege_favour_strong_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = liege_favour_strong_hook target = scope:target }
		}
		else_if = {
			limit = {
				has_hook_of_type = {
					target = scope:target
					type = saved_my_life_hook
				}
			}
			remove_hook = { target = scope:target }
			add_hook = { type = saved_my_life_hook target = scope:target }
		}
		else = {
			remove_hook = { target = scope:target }
			add_hook = { type = strong_favor_hook target = scope:target }
		}
	}
}

# Clear Fabricate Hook scheme CD against a saved target (after scheme end applies CD).
# Expects this = owner; var:ssc_hook_cd_clear_target set.
ssc_clear_fabricate_hook_cooldown_effect = {
	if = {
		limit = {
			ssc_player_bypasses_hook_cooldowns = yes
			exists = var:ssc_hook_cd_clear_target
			var:ssc_hook_cd_clear_target = { is_alive = yes }
		}
		remove_scheme_cooldown_against = {
			scheme = fabricate_hook
			target = var:ssc_hook_cd_clear_target
		}
		remove_variable = ssc_hook_cd_clear_target
	}
}

# Queue delayed clear; root should be scheme owner with scope:target present.
ssc_queue_fabricate_hook_cooldown_clear_effect = {
	if = {
		limit = {
			ssc_player_bypasses_hook_cooldowns = yes
			exists = scope:target
			scope:target = { is_alive = yes }
		}
		set_variable = {
			name = ssc_hook_cd_clear_target
			value = scope:target
		}
		trigger_event = {
			id = ssc_events.4001
			days = 1
		}
	}
}
