			# Former decision is_shown
			NOT = { has_game_rule = ssc_rule_granting_disabled }
			ssc_character_is_supreme_sovereign = yes
			# Former is_valid

				text = ssc_tt_actor_may_demand_strong_hook
				ssc_can_demand_strong_hook_actor = yes
			}

				text = ssc_tt_has_hookable_vassals

					save_temporary_scope_as = ssc_hook_candidate

					NOT = { this = root }
					root = {
						can_add_hook = {
							type = strong_favor_hook
							target = scope:ssc_hook_candidate
						}
					}
				}
			}
		}
		custom_tooltip = ssc_hook_all_vassals_decision_effect_tt
		ssc_hook_all_direct_vassals_effect = yes
	}
