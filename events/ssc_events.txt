# Artifact picker (player, on scheme start)
ssc_events.2010 = {
	type = character_event
	title = ssc_events.2010.t
	desc = ssc_events.2010.desc
	theme = intrigue
	left_portrait = {
		character = root
		animation = scheme
	}
	lower_right_portrait = scope:target
	artifact = {
		target = scope:artifact
		position = lower_center_portrait
		trigger = { exists = scope:artifact }
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:scheme } }
			random_scheme = {
				limit = { scheme_type = ssc_steal_artifact }
				save_scope_as = scheme
			}
		}
		if = {
			limit = {
				exists = scope:scheme
				NOT = { exists = scope:artifact }
			}
			scope:scheme = {
				if = {
					limit = { exists = var:target_artifact }
					var:target_artifact = { save_scope_as = artifact }
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:target } }
			scope:scheme = {
				scheme_target_character = { save_scope_as = target }
			}
		}
	}

	option = {
		name = ssc_events.2010.a
		trigger = { exists = scope:artifact }
		custom_tooltip = ssc_events.2010.a.tt
	}

	option = {
		name = ssc_events.2010.b
		trigger = {
			exists = scope:target
			scope:target = {
				any_character_artifact = {
					rarity = illustrious
				}
			}
		}
		custom_tooltip = ssc_events.2010.b.tt
		scope:target = {
			random_character_artifact = {
				limit = { rarity = illustrious }
				save_scope_as = artifact
			}
		}
		if = {
			limit = {
				exists = scope:artifact
				exists = scope:scheme
			}
			scope:artifact = {
				ssc_set_scheme_artifact_target_effect = yes
			}
		}
	}

	option = {
		name = ssc_events.2010.c
		custom_tooltip = ssc_events.2010.c.tt
		scope:target = {
			random_character_artifact = {
				save_scope_as = artifact
			}
		}
		if = {
			limit = {
				exists = scope:artifact
				exists = scope:scheme
			}
			scope:artifact = {
				ssc_set_scheme_artifact_target_effect = yes
			}
		}
	}
}

# M5 — Force Contract spouse picker (on scheme start, player)
ssc_events.5101 = {
	type = character_event
	title = ssc_events.5101.t
	desc = ssc_events.5101.desc
	theme = intrigue
	left_portrait = {
		character = root
		animation = scheme
	}
	lower_right_portrait = scope:target

	immediate = {
		if = {
			limit = { NOT = { exists = scope:scheme } }
			random_scheme = {
				limit = { scheme_type = ssc_force_contract }
				save_scope_as = scheme
			}
		}
		if = {
			limit = { NOT = { exists = scope:target } }
			random_scheme = {
				limit = { scheme_type = ssc_force_contract }
				scheme_target_character = { save_scope_as = target }
				save_scope_as = scheme
			}
		}
	}

	option = {
		name = ssc_events.5101.a
		custom_tooltip = ssc_events.5101.a.tt
		trigger = {
			is_married = no
			exists = scope:target
			can_marry_character = scope:target
			scope:target = {
				is_married = no
				can_marry_character = root
			}
		}
		if = {
			limit = { exists = scope:scheme }
			scope:scheme = {
				set_variable = {
					name = ssc_contract_spouse
					value = root
				}
			}
		}
	}

	option = {
		name = ssc_events.5101.b
		custom_tooltip = ssc_events.5101.b.tt
		random_courtier = {
			limit = {
				is_alive = yes
				is_adult = yes
				is_married = no
				exists = scope:target
				can_marry_character = scope:target
				scope:target = { can_marry_character = prev }
			}
			save_scope_as = ssc_contract_spouse
		}
		if = {
			limit = {
				exists = scope:scheme
				exists = scope:ssc_contract_spouse
			}
			scope:scheme = {
				set_variable = {
					name = ssc_contract_spouse
					value = scope:ssc_contract_spouse
				}
			}
		}
	}

	option = {
		name = ssc_events.5101.c
		custom_tooltip = ssc_events.5101.c.tt
		# Leave unset — success path auto-picks best eligible
	}
}
