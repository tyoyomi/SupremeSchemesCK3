# Hook Economy Board (M6)
ssc_settings.1101 = {
	type = character_event
	title = ssc_settings.1101.t
	theme = intrigue

	desc = {
		desc = ssc_settings.1101.desc_intro
		desc = ssc_settings.1001.desc_blank
		desc = ssc_settings.1101.desc_verbs
		desc = ssc_settings.1001.desc_blank
		desc = ssc_settings.1101.desc_influence
	}

	immediate = {
		ssc_ensure_schemer_defaults_effect = yes
	}

	option = {
		name = ssc_settings.1101.opt.buy_all_pact
		custom_tooltip = ssc_settings.1101.tt.buy_all_pact
		trigger = {
			ssc_can_use_hook_board = yes
			ssc_su_soft_available = yes
			is_dynast = yes
			any_dynasty_member = {
				is_alive = yes
				NOT = { this = root }
				dynasty = root.dynasty
				opinion = {
					target = root
					value < 25
				}
			}
		}
		ssc_he_buy_all_pact_approvals_effect = yes
		trigger_event = ssc_settings.1101
	}

	option = {
		name = ssc_settings.1101.opt.silence_faction
		custom_tooltip = ssc_settings.1101.tt.silence_faction
		trigger = {
			ssc_can_use_hook_board = yes
			any_vassal = {
				exists = faction
				root = { has_strong_usable_hook = prev }
			}
		}
		random_vassal = {
			limit = {
				exists = faction
				root = { has_strong_usable_hook = prev }
			}
			save_scope_as = recipient
		}
		ssc_he_silence_faction_effect = yes
		trigger_event = ssc_settings.1101
	}

	option = {
		name = ssc_settings.1101.opt.hint_interactions
		custom_tooltip = ssc_settings.1101.tt.hint_interactions
	}

	option = {
		name = ssc_settings.opt.back_to_hub
		custom_tooltip = ssc_settings.tt.back_to_hub
		trigger_event = ssc_settings.1001
	}

	option = {
		name = ssc_settings.1001.close
		custom_tooltip = ssc_settings.tt.close
	}
}
